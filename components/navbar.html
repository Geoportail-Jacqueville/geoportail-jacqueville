FONCTIONS UTILITAIRES ----- function injectHTML(placeholderId, html) { const el = document.getElementById(placeholderId); if (el) el.innerHTML = html; } async function loadComponent(url, placeholderId, fallbackHtml) { try { const response = await fetch(url); if (!response.ok) throw new Error('HTTP ' + response.status); const html = await response.text(); injectHTML(placeholderId, html); } catch (err) { console.warn('Fallback pour ' + placeholderId + ' (' + err.message + ')'); injectHTML(placeholderId, fallbackHtml); } } // ----- 4. INITIALISATION DE LA NAVBAR (correction des liens) ----- function initNavbar() { const isInPagesFolder = window.location.pathname.includes('/pages/'); // Correction des chemins de tous les liens de la navbar document.querySelectorAll('.gp-nav a[href]').forEach(link => { const href = link.getAttribute('href'); if (!href || href.startsWith('http') || href.startsWith('#') || href.startsWith('mailto:') || href.startsWith('tel:')) return; if (isInPagesFolder) { // On est dans /pages/ : tous les liens relatifs doivent remonter d'un niveau if (href === 'index.html') { link.setAttribute('href', '../index.html'); } else if (href.startsWith('pages/')) { link.setAttribute('href', '../' + href); } else if (!href.startsWith('../')) { link.setAttribute('href', '../' + href); } } else { // On est à la racine : les liens vers les pages doivent être préfixés par "pages/" if (href !== 'index.html' && !href.startsWith('pages/') && !href.startsWith('http')) { link.setAttribute('href', 'pages/' + href); } } }); // Burger menu const burger = document.querySelector('.gp-nav-burger'); const mobileMenu = document.querySelector('.gp-nav-mobile'); if (burger && mobileMenu) { burger.addEventListener('click', () => { const isOpen = burger.classList.toggle('open'); mobileMenu.classList.toggle('open', isOpen); burger.setAttribute('aria-expanded', isOpen); mobileMenu.setAttribute('aria-hidden', !isOpen); document.body.style.overflow = isOpen ? 'hidden' : ''; }); mobileMenu.querySelectorAll('a').forEach(link => { link.addEventListener('click', () => { burger.classList.remove('open'); mobileMenu.classList.remove('open'); burger.setAttribute('aria-expanded', 'false'); mobileMenu.setAttribute('aria-hidden', 'true'); document.body.style.overflow = ''; }); }); } // Lien actif (basé sur data-page) const path = window.location.pathname; let current = path.split('/').pop().replace('.html', ''); if (!current || current === 'index') current = 'accueil'; document.querySelectorAll('[data-page]').forEach(link => { if (link.dataset.page === current) link.classList.add('active'); }); } // ----- 5. INITIALISATION DU FOOTER ----- function initFooter() { // Année dynamique const yearSpan = document.querySelector('.js-current-year'); if (yearSpan) yearSpan.textContent = new Date().getFullYear(); // Bouton retour en haut const backBtn = document.querySelector('.js-back-to-top'); if (backBtn) { backBtn.addEventListener('click', e => { e.preventDefault(); window.scrollTo({ top: 0, behavior: 'smooth' }); }); window.addEventListener('scroll', () => { backBtn.classList.toggle('visible', window.scrollY > 400); }); } } // ----- 6. LANCEMENT ----- async function init() { const navbarURL = basePath + 'components/navbar.html'; const footerURL = basePath + 'components/footer.html'; await Promise.all([ loadComponent(navbarURL, 'navbar-placeholder', fallbackNavbar), loadComponent(footerURL, 'footer-placeholder', fallbackFooter) ]); // Laisser le DOM se mettre à jour setTimeout(() => { initNavbar(); initFooter(); }, 10); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', init); } else { init(); } })();
